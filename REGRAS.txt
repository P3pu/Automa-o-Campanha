    // 1ª CONDIÇÃO (Grupos/Company/Saude Corportativa): Retorna "Company" se houver "Grupos" em qualquer uma das 3 colunas.
    CONTAINSSTRING(Tb_Base[Categoria], "Grupos") || 
    CONTAINSSTRING(Tb_Base[Etiqueta], "GRUPOS") ||

    CONTAINSSTRING(Tb_Base[Local Inscrição], "Balcão") ||
